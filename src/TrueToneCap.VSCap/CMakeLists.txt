cmake_minimum_required(VERSION 3.25)
project(TrueToneCapVSCap LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
qt_add_executable(ttc-vscap main.cpp upstream/src/cap/DesktopCapture.cpp)
target_compile_definitions(ttc-vscap PRIVATE VSR_VERSION="1.0.10")
target_link_libraries(ttc-vscap PRIVATE Qt6::Widgets user32 d3d11 dxgi)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(vscap-workflow-tests upstream/tests/TestCapWorkflow.cpp upstream/src/cap/DesktopCapture.cpp)
    target_link_libraries(vscap-workflow-tests PRIVATE Qt6::Widgets Qt6::Test user32 d3d11 dxgi)
    add_test(NAME vscap-workflow COMMAND vscap-workflow-tests)
    qt_add_executable(vscap-bridge-tests tests/BridgeTests.cpp upstream/src/cap/DesktopCapture.cpp)
    target_link_libraries(vscap-bridge-tests PRIVATE Qt6::Widgets Qt6::Test user32 d3d11 dxgi)
    add_test(NAME vscap-bridge COMMAND vscap-bridge-tests)
endif()
